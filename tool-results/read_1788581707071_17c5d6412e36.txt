   472→// ---------------------------------------------------------------------------
   473→// Stage 6 — Safety Check (directive §2/§46 "Check Before You Deal").
   474→// A signed-in VERIFIER runs a sanitized assessment on a subject via:
   475→//   HANDLE        — the subject's @handle (requires their standing safety
   476→//                   consent; anti-enumeration: "no such handle" and "checks
   477→//                   disabled" are byte-identical responses)
   478→//   PHONE         — consent-gated peppered-hash match against the subject's
   479→//                   PHONE identifier (the raw phone is hashed and discarded;
   480→//                   it is NEVER stored, logged or echoed)
   481→//   TRUST_LINK/QR — an active Stage 5 share token / QR Trust Card payload
   482→//                   (token scopes govern; the view is counted and receipted
   483→//                   with the verifier's NAME — a named check)
   484→// Every completed check writes a SafetyCheck row (the snapshot the verifier
   485→// saw), a receipt + notification for the subject, and an audit event.
   486→// The assessment NEVER says "this person is safe" (directive §50).
   487→// ---------------------------------------------------------------------------
   488→
   489→model SafetyCheck {
   490→  id           String    @id @default(cuid())
   491→  verifierId   String
   492→  subjectId    String
   493→  method       String // HANDLE | PHONE | TRUST_LINK | QR
   494→  assessment   String // JSON snapshot of exactly what the verifier saw (sanitized, no raw identifiers)
   495→  consentId    String? // standing SAFETY_CHECK consent that authorized a HANDLE/PHONE check
   496→  shareTokenId String? // governing token for TRUST_LINK/QR checks
   497→  ipHash       String? // salted — abuse dedupe only
   498→  createdAt    DateTime  @default(now())
   499→
   500→  verifier UserAccount @relation("SafetyChecksMade", fields: [verifierId], references: [id], onDelete: Cascade)
   501→  subject  UserAccount @relation("SafetyChecksReceived", fields: [subjectId], references: [id], onDelete: Cascade)
   502→
   503→  @@index([verifierId, createdAt])
   504→  @@index([subjectId, createdAt])
   505→}

   510→// accepts (a scoped trust link is minted and the assessment runs) or
   511→// declines. Requests expire after 7 days; one pending request per pair.
   512→// ---------------------------------------------------------------------------
   513→
   514→model TrustRequest {
   515→  id           String    @id @default(cuid())
   516→  verifierId   String
   517→  subjectId    String
   518→  status       String    @default("PENDING") // PENDING | ACCEPTED | DECLINED | EXPIRED
   519→  message      String // short preset ask — no free-text PII surface
   520→  shareTokenId String? // minted on ACCEPT
   521→  expiresAt    DateTime
   522→  respondedAt  DateTime?
   523→  createdAt    DateTime  @default(now())
   524→
   525→  verifier UserAccount @relation("TrustRequestsSent", fields: [verifierId], references: [id], onDelete: Cascade)
   526→  subject  UserAccount @relation("TrustRequestsReceived", fields: [subjectId], references: [id], onDelete: Cascade)
   527→
   528→  @@index([subjectId, status])
   529→  @@index([verifierId, createdAt])
   530→}

   549→// ---------------------------------------------------------------------------
   550→// Stage 7 — Reputation: Flags with evidence (directive §29 flags/flag_evidence).
   551→// A flag is a REPORT TO THE PLATFORM about a member's conduct — never a
   552→// public accusation: nothing about a flag is visible to anyone except the
   553→// reporter, the subject and human reviewers. Anti-gaming by design
   554→// (enforced in reputation-service):
   555→//   * the reporter must hold a live L2+ Trust Identity (a verified human)
   556→//   * one OPEN/UNDER_REVIEW flag per reporter→subject pair (409 duplicate)
   557→//   * at most 3 flags per rolling 7-day window per reporter (429 window)
   558→//   * at least one evidence item is mandatory at submission (422 zod)
   559→//   * reporter identity is MASKED to the subject (retaliation shield);
   560→//     the full record remains available via NDPA DSR export
   561→// ---------------------------------------------------------------------------
   562→
   563→model Flag {
   564→  id                 String    @id @default(cuid())
   565→  reporterId         String
   566→  subjectId          String
   567→  category           String // FRAUD | IMPERSONATION | NON_PAYMENT | SCAM | HARASSMENT | OTHER
   568→  description        String // the reporter's claim — shown to subject + reviewer
   569→  status             String    @default("OPEN") // OPEN | UNDER_REVIEW | RESOLVED_CONFIRMED | RESOLVED_UNFOUNDED | RESOLVED_DISMISSED | WITHDRAWN
   570→  evidenceCount      Int       @default(0)
   571→  subjectRespondedAt DateTime?
   572→  createdAt          DateTime  @default(now())
   573→  updatedAt          DateTime  @updatedAt
   574→
   575→  reporter   UserAccount     @relation("FlagsFiled", fields: [reporterId], references: [id], onDelete: Cascade)
   576→  subject    UserAccount     @relation("FlagsReceived", fields: [subjectId], references: [id], onDelete: Cascade)
   577→  evidence   FlagEvidence[]
   578→  resolution FlagResolution?
   579→  appeal     FlagAppeal?
   580→
   581→  @@index([subjectId, status])
   582→  @@index([reporterId, createdAt])
   583→}

   585→// Evidence items attached by either party. Content is their own statement or

   595→  content       String
   596→  createdAt     DateTime @default(now())
   597→
   598→  flag Flag @relation(fields: [flagId], references: [id], onDelete: Cascade)
   599→
   600→  @@index([flagId])
   601→}

   603→// Human-reviewed resolution (directive §33 — no automated significant
   604→// decisions in Stage 7). A REVIEWER decides CONFIRMED / UNFOUNDED / DISMISSED
   605→// with a published rationale. CONFIRMED sets fraudSignal=true and feeds the
   606→// score engine's Confirmed Risk component (−25 per confirmed flag, cap −50).
   607→
   608→model FlagResolution {
   609→  id          String   @id @default(cuid())
   610→  flagId      String   @unique
   611→  reviewerId  String
   612→  outcome     String // CONFIRMED | UNFOUNDED | DISMISSED
   613→  rationale   String // published to subject + reporter
   614→  fraudSignal Boolean  @default(false) // true when outcome = CONFIRMED
   615→  decidedAt   DateTime @default(now())
   616→
   617→  reviewer UserAccount @relation("FlagResolutions", fields: [reviewerId], references: [id], onDelete: Cascade)
   618→  flag     Flag        @relation(fields: [flagId], references: [id], onDelete: Cascade)
   619→
   620→  @@index([reviewerId])
   621→}

   623→// Subject appeal of a CONFIRMED resolution — one per flag, 14-day window
   624→// from the decision. UPHELD keeps the outcome; OVERTURNED flips the flag to
   625→// RESOLVED_UNFOUNDED and restores the score (NDPA §37 redress path).

   627→model FlagAppeal {
   628→  id           String    @id @default(cuid())
   629→  flagId       String    @unique
   630→  appellantId  String // the flag subject
   631→  reason       String
   632→  status       String    @default("PENDING") // PENDING | UPHELD | OVERTURNED
   633→  reviewerId   String?
   634→  decisionNote String?
   635→  decidedAt    DateTime?
   636→  createdAt    DateTime  @default(now())
   637→
   638→  appellant UserAccount @relation("FlagAppeals", fields: [appellantId], references: [id], onDelete: Cascade)
   639→  flag      Flag        @relation(fields: [flagId], references: [id], onDelete: Cascade)
   640→
   641→  @@index([status])
   642→}
