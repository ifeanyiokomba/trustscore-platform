     1→// TrustScore Stage 5–8 — TrustScore engine (directive §35/§36 score contract).
     2→//
     3→// Formula (directive §36):
     4→//   score = Identity Assurance + Verified Reputation + Verified Credentials
     5→//           + Resolution History − Confirmed Risk
     6→//
     7→// Stage 8 (Trust Engine): every budget/threshold/window now lives in a
     8→// versioned ScoringPolicy row (rules-first, append-only). This module keeps
     9→// the §36 math but reads its numbers from the ACTIVE policy — policy v1 is
    10→// byte-for-byte the budgets shipped in Stage 5–7, so scores are unchanged
    11→// until a new policy version is deliberately activated.
    12→//
    13→// Output contract: Score (0–100) + Confidence (0–100) + Risk Band
    14→// (LOW/MEDIUM/HIGH) + Status (NEW/VERIFIED/ESTABLISHED/CAUTION/HIGH_RISK/
    15→// REVIEW_REQUIRED) + Explanation[] + Freshness (computedAt → expiresAt) +
    16→// State (ACTIVE/STALE/FROZEN/RETIRED) + policy version.
    17→//
    18→// Snapshots are immutable rows: a new one is written when inputs change
    19→// (inputsHash incl. policy hash) or the TTL lapses. Old snapshots beyond the
    20→// latest 20 are pruned. The engine NEVER says "this person is safe"
    21→// (directive §50) — the public card language is locked to "No confirmed
    22→// adverse signals found".
    23→// While the latest snapshot is FROZEN (appeal pending), getScoreSnapshot
    24→// serves it verbatim — no recompute, no new rows (fairness guarantee).
    25→

    49→export interface ComputedScore {
    50→  status: ScoreStatus;
    51→  score: number;
    52→  confidence: number;
    53→  riskBand: "LOW" | "MEDIUM" | "HIGH";
    54→  components: ScoreComponent[];
    55→  explanation: string[];
    56→  inputsHash: string;
    57→  trigger: "INITIAL" | "MATERIAL_CHANGE" | "PERIODIC";
    58→  computedAt: string; // ISO
    59→  expiresAt: string; // ISO
    60→  policyVersion: number; // Stage 8: the rules that produced this score
    61→}

    63→const ASSURANCE_BASE: Record<number, number> = { 0: 0, 1: 20, 2: 34, 3: 47, 4: 60 };
    64→
    65→// Legacy constant kept for imports elsewhere (Stage 5–7 tests/docs) — the
    66→// live TTL now comes from the ACTIVE policy's snapshotTtlHours.
    67→export const SNAPSHOT_TTL_MS = 24 * 60 * 60 * 1000;

    69→interface ScoreInputs {
    70→  identity: {
    71→    status: string;
    72→    assuranceLevel: number;
    73→    govExpiresAt: Date | null;
    74→    verifiedAt: Date | null;
    75→  } | null;
    76→  activeEvidence: {
    77→    id: string;
    78→    type: string;
    79→    status: string;
    80→    confidence: number;
    81→    expiresAt: Date | null;
    82→  }[];
    83→  credentials: {
    84→    id: string;
    85→    type: string;
    86→    status: string;
    87→    expiresAt: Date | null;
    88→    manualRevoked: boolean;
    89→  }[];
    90→  // Stage 7 — reputation inputs (computed here so the engine owns its inputs)
    91→  verifiedInteractions: number; // distinct L2+ verifiers, consent-backed checks, 90d
    92→  clearedFlags: number; // human-reviewed UNFOUNDED / DISMISSED / OVERTURNED
    93→  confirmedFlags: number; // human-confirmed flags (appeals upheld or none)
    94→}
    95→
    96→// Verified-interaction horizon: checks older than the policy window stop
    97→// counting — reputation must be earned continuously, not banked once.
    98→const INTERACTION_WINDOW_MS = 90 * 24 * 60 * 60 * 1000;
    99→
   100→async function collectReputationInputs(
   101→  userId: string,
   102→  interactionWindowDays?: number
   103→): Promise<{
   104→  verifiedInteractions: number;
   105→  clearedFlags: number;
   106→  confirmedFlags: number;
   107→}> {
   108→  const windowMs =
   109→    interactionWindowDays && interactionWindowDays > 0
   110→      ? interactionWindowDays * 24 * 60 * 60 * 1000
   111→      : INTERACTION_WINDOW_MS;
   112→  const flags = await db.flag.findMany({
   113→    where: { subjectId: userId },
   114→    select: { status: true },
   115→  });
   116→  const confirmedFlags = flags.filter((f) => f.status === "RESOLVED_CONFIRMED").length;
   117→  const clearedFlags = flags.filter((f) =>
   118→    ["RESOLVED_UNFOUNDED", "RESOLVED_DISMISSED"].includes(f.status)
   119→  ).length;
   120→
   121→  // Verified interactions: distinct verifiers who ran a CONSENT-BACKED check
   122→  // (handle / trust link / QR — never the anonymous phone probe) on this
   123→  // member within the window, and who currently hold L2+.
   124→  const checks = await db.safetyCheck.findMany({
   125→    where: {
   126→      subjectId: userId,
   127→      method: { in: ["HANDLE", "TRUST_LINK", "QR"] },
   128→      createdAt: { gte: new Date(Date.now() - windowMs) },
   129→    },
   130→    select: { verifierId: true },
   131→  });
   132→  const verifierIds = [...new Set(checks.map((c) => c.verifierId))].filter(
   133→    (id) => id !== userId
   134→  );
   135→  let verifiedInteractions = 0;
   136→  if (verifierIds.length > 0) {
   137→    const identities = await db.trustIdentity.findMany({
   138→      where: { userId: { in: verifierIds } },
   139→      select: { userId: true, status: true, assuranceLevel: true, expiresAt: true },
   140→    });
   141→    const now = Date.now();
   142→    for (const id of identities) {
   143→      const live = id.status === "VERIFIED" && (id.expiresAt?.getTime() ?? 0) > now;
   144→      if (live && id.assuranceLevel >= 2) verifiedInteractions += 1;
   145→    }
   146→  }
   147→  return { verifiedInteractions, clearedFlags, confirmedFlags };
   148→}

   150→async function collectInputs(
   151→  userId: string,
   152→  interactionWindowDays?: number
   153→): Promise<ScoreInputs> {
   154→  const identity = await db.trustIdentity.findUnique({
   155→    where: { userId },
   156→    select: {
   157→      status: true,
   158→      assuranceLevel: true,
   159→      expiresAt: true,
   160→      verifiedAt: true,
   161→      evidence: {
   162→        where: { status: "ACTIVE" },
   163→        select: { id: true, type: true, confidence: true, expiresAt: true },
   164→      },
   165→    },
   166→  });
   167→  const credentials = await db.credential.findMany({
   168→    where: { userId },
   169→    select: { id: true, type: true, status: true, expiresAt: true, manualRevoked: true },
   170→  });
   171→  const reputation = await collectReputationInputs(userId, interactionWindowDays);
   172→  return {
   173→    identity: identity
   174→      ? {
   175→          status: identity.status,
   176→          assuranceLevel: identity.assuranceLevel,
   177→          govExpiresAt: identity.expiresAt,
   178→          verifiedAt: identity.verifiedAt,
   179→        }
   180→      : null,
   181→    activeEvidence: (identity?.evidence ?? []).map((e) => ({
   182→      id: e.id,
   183→      type: e.type,
   184→      status: "ACTIVE" as const,
   185→      confidence: e.confidence,
   186→      expiresAt: e.expiresAt,
   187→    })),
   188→    credentials,
   189→    verifiedInteractions: reputation.verifiedInteractions,
   190→    clearedFlags: reputation.clearedFlags,
   191→    confirmedFlags: reputation.confirmedFlags,
   192→  };
   193→}

   195→function inputsHashFor(inputs: ScoreInputs): string {
   196→  const canonical = JSON.stringify({
   197→    i: inputs.identity
   198→      ? [inputs.identity.status, inputs.identity.assuranceLevel, inputs.identity.govExpiresAt?.getTime() ?? null]
   199→      : null,
   200→    e: inputs.activeEvidence
   201→      .map((e) => [e.id, e.status ?? "ACTIVE", e.expiresAt?.getTime() ?? null])
   202→      .sort(),
   203→    c: inputs.credentials.map((c) => [c.type, c.status, c.manualRevoked]).sort(),
   204→    r: [inputs.confirmedFlags, inputs.clearedFlags, inputs.verifiedInteractions],
   205→  });
   206→  return sha256Hex(canonical);
   207→}

   209→function clamp(n: number, min: number, max: number): number {
   210→  return Math.max(min, Math.min(max, n));
   211→}

   213→export function computeScoreFromInputs(
   214→  inputs: ScoreInputs,
   215→  trigger: ComputedScore["trigger"],
   216→  rules: PolicyRules,
   217→  ttlMs: number
   218→): ComputedScore {
   219→  const now = Date.now();
   220→  const identityLive =
   221→    inputs.identity !== null &&
   222→    inputs.identity.status === "VERIFIED" &&
   223→    (inputs.identity.govExpiresAt?.getTime() ?? 0) > now;
   224→  const identityStale =
   225→    inputs.identity !== null &&
   226→    inputs.identity.status === "VERIFIED" &&
   227→    (inputs.identity.govExpiresAt?.getTime() ?? 0) <= now;
   228→  const level = identityLive ? inputs.identity!.assuranceLevel : 0;
   229→  const ASSURANCE = { 0: rules.assuranceBase[0], 1: rules.assuranceBase[1], 2: rules.assuranceBase[2], 3: rules.assuranceBase[3], 4: rules.assuranceBase[4] };
   230→
   231→  // --- Identity Assurance (budget: assuranceBase[4]) ----------------------
   232→  const govDaysLeft = identityLive
   233→    ? Math.max(0, (inputs.identity!.govExpiresAt!.getTime() - now) / 86_400_000)
   234→    : 0;
   235→  const freshnessFactor = !identityLive
   236→    ? 0
   237→    : govDaysLeft >= rules.freshnessFullDays
   238→      ? 1
   239→      : govDaysLeft >= 1
   240→        ? Math.max(rules.freshnessMinFactor, 0.9)
   241→        : rules.freshnessMinFactor;
   242→  const assuranceValue = Math.round((ASSURANCE[clamp(level, 0, 4)] ?? 0) * freshnessFactor);
   243→
   244→  // --- Verified Credentials (policy: credentialPoints/credentialMax) ------
   245→  const freshCredentials = inputs.credentials.filter(
   246→    (c) => c.status === "ACTIVE" && (c.expiresAt?.getTime() ?? Infinity) > now
   247→  );
   248→  const credentialValue = clamp(freshCredentials.length * rules.credentialPoints, 0, rules.credentialMax);
   249→
   250→  // --- Verified Reputation / Resolution History (policy-driven) ----------
   251→  const reputationValue = Math.min(rules.interactionMax, inputs.verifiedInteractions) * rules.interactionPoints;
   252→  const resolutionValue = Math.min(rules.clearedMax, inputs.clearedFlags) * rules.clearedPoints;
   253→
   254→  // --- Confirmed Risk (policy: riskPenaltyPer/riskMaxPenalty) ------------
   255→  const riskPenalty = Math.min(
   256→    Math.floor(rules.riskMaxPenalty / Math.max(1, rules.riskPenaltyPer)),
   257→    inputs.confirmedFlags
   258→  ) * rules.riskPenaltyPer;
   259→
   260→  const score = clamp(
   261→    assuranceValue + credentialValue + reputationValue + resolutionValue - riskPenalty,
   262→    0,
   263→    100
   264→  );
   265→
   266→  // --- Confidence -----------------------------------------------------------
   267→  const freshEvidence = inputs.activeEvidence.filter(
   268→    (e) => (e.expiresAt?.getTime() ?? Infinity) > now
   269→  );
   270→  const allFresh =
   271→    freshEvidence.length === inputs.activeEvidence.length && identityLive;
   272→  const confidence = !inputs.identity
   273→    ? 15
   274→    : clamp(40 + Math.min(40, freshEvidence.length * 10) + (allFresh ? 20 : 0), 0, 100);
   275→
   276→  // --- Status + risk band ---------------------------------------------------
   277→  let status: ScoreStatus;
   278→  let riskBand: "LOW" | "MEDIUM" | "HIGH";
   279→  if (inputs.confirmedFlags > 0) {
   280→    status = inputs.confirmedFlags >= rules.riskHighAt ? "HIGH_RISK" : "REVIEW_REQUIRED";
   281→    riskBand = "HIGH";
   282→  } else if (identityStale) {
   283→    status = "CAUTION";
   284→    riskBand = "MEDIUM";
   285→  } else if (!inputs.identity || inputs.identity.status === "NONE" || level === 0) {
   286→    status = "NEW";
   287→    riskBand = "MEDIUM";
   288→  } else if (level >= rules.establishedMinLevel) {
   289→    status = "ESTABLISHED";
   290→    riskBand = "LOW";
   291→  } else {
   292→    status = "VERIFIED";
   293→    riskBand = "LOW";
   294→  }
   295→
   296→  // --- Explanation (NDPA §37 — meaningful, per-component) -------------------
   297→  const explanation: string[] = [];
   298→  if (!inputs.identity) {
   299→    explanation.push(
   300→      "No Trust Identity yet — this account is unverified, so the score starts at zero."
   301→    );
   302→  } else {
   303→    explanation.push(
   304→      `Identity Assurance ${assuranceValue}/${ASSURANCE[4]}: government record ${identityLive ? "verified" : identityStale ? "past its freshness horizon" : inputs.identity.status.toLowerCase()}, assurance level L${level}${identityLive ? `, ${Math.round(govDaysLeft)} days of freshness left` : ""}.`
   305→    );
   306→    explanation.push(
   307→      freshCredentials.length > 0
   308→        ? `Verified Credentials ${credentialValue}/${rules.credentialMax}: ${freshCredentials.length} active credential${freshCredentials.length === 1 ? "" : "s"} backed by live evidence (${freshCredentials.map((c) => c.type.replace("_VERIFIED", "").replace(/_/g, " ").toLowerCase()).join(", ")}).`
   309→        : `Verified Credentials 0/${rules.credentialMax}: no active credentials — verify signals to earn them.`
   310→    );
   311→    explanation.push(
   312→      inputs.verifiedInteractions > 0
   313→        ? `Verified Reputation ${reputationValue}/${rules.interactionMax * rules.interactionPoints}: ${inputs.verifiedInteractions} verified interaction${inputs.verifiedInteractions === 1 ? "" : "s"} with distinct verified members (consent-backed checks in the last ${rules.interactionWindowDays} days, +${rules.interactionPoints} each, capped at ${rules.interactionMax}).`
   314→        : `Verified Reputation 0/${rules.interactionMax * rules.interactionPoints}: no consent-backed checks by verified members in the last ${rules.interactionWindowDays} days — an honest zero, not a negative signal.`
   315→    );
   316→    explanation.push(
   317→      inputs.clearedFlags > 0
   318→        ? `Resolution History ${resolutionValue}/${rules.clearedMax * rules.clearedPoints}: ${inputs.clearedFlags} flag${inputs.clearedFlags === 1 ? " was" : "s were"} raised against you and cleared by human review (+${rules.clearedPoints} each, capped at ${rules.clearedMax}).`
   319→        : `Resolution History 0/${rules.clearedMax * rules.clearedPoints}: no human-reviewed flag resolutions on record yet — an honest zero, not a negative signal.`
   320→    );
   321→    explanation.push(
   322→      riskPenalty > 0
   323→        ? `Confirmed Risk −${riskPenalty}: ${inputs.confirmedFlags} confirmed flag${inputs.confirmedFlags === 1 ? "" : "s"} on record after human review. You can appeal a confirmation for 14 days — while an appeal is pending, your score is frozen so it cannot move against you.`
   324→        : `Confirmed Risk −0: no confirmed adverse signals found. This is a statement about recorded evidence only — never a guarantee that a person is safe to deal with.`
   325→    );
   326→    explanation.push(
   327→      `Confidence ${confidence}/100: based on ${freshEvidence.length} fresh evidence record${freshEvidence.length === 1 ? "" : "s"}${allFresh ? " with all signals inside their freshness horizons" : " (some signals are stale)"}.`
   328→    );
   329→  }
   330→  explanation.push(
   331→    "Under NDPA §37 you can request human review of this decision; confirmed flags can be appealed for 14 days after the reviewer's decision."
   332→  );
   333→
   334→  return {
   335→    status,
   336→    score,
   337→    confidence,
   338→    riskBand,
   339→    components: [
   340→      {
   341→        key: "identityAssurance",
   342→        label: "Identity Assurance",
   343→        value: assuranceValue,
   344→        max: ASSURANCE[4],
   345→        note: identityLive
   346→          ? `Government-verified, L${level}${freshnessFactor < 1 ? ", freshness-scaled" : ""}`
   347→          : "No live government verification",
   348→      },
   349→      {
   350→        key: "verifiedCredentials",
   351→        label: "Verified Credentials",
   352→        value: credentialValue,
   353→        max: rules.credentialMax,
   354→        note: `${freshCredentials.length} active, evidence-backed (+${rules.credentialPoints} each)`,
   355→      },
   356→      {
   357→        key: "verifiedReputation",
   358→        label: "Verified Reputation",
   359→        value: reputationValue,
   360→        max: rules.interactionMax * rules.interactionPoints,
   361→        note:
   362→          inputs.verifiedInteractions > 0
   363→            ? `${inputs.verifiedInteractions} verified interaction${inputs.verifiedInteractions === 1 ? "" : "s"} (${rules.interactionWindowDays}-day window)`
   364→            : `No verified interactions in the last ${rules.interactionWindowDays} days`,
   365→      },
   366→      {
   367→        key: "resolutionHistory",
   368→        label: "Resolution History",
   369→        value: resolutionValue,
   370→        max: rules.clearedMax * rules.clearedPoints,
   371→        note:
   372→          inputs.clearedFlags > 0
   373→            ? `${inputs.clearedFlags} flag${inputs.clearedFlags === 1 ? "" : "s"} cleared by human review`
   374→            : "No cleared flags on record yet",
   375→      },
   376→      {
   377→        key: "confirmedRisk",
   378→        label: "Confirmed Risk",
   379→        value: -riskPenalty,
   380→        max: 0,
   381→        note:
   382→          inputs.confirmedFlags > 0
   383→            ? `${inputs.confirmedFlags} confirmed flag${inputs.confirmedFlags === 1 ? "" : "s"} (human-reviewed, appealable; −${rules.riskPenaltyPer} each, cap −${rules.riskMaxPenalty})`
   384→            : `No confirmed flags on record (−${rules.riskPenaltyPer} each when confirmed)`,
   385→      },
   386→    ],
   387→    explanation,
   388→    // Combined hash: inputs + the policy rules that interpreted them —
   389→    // a policy change invalidates every cached snapshot (Stage 8).
   390→    inputsHash: sha256Hex(inputsHashFor(inputs) + policyRulesHash(rules)),
   391→    trigger,
   392→    computedAt: new Date(now).toISOString(),
   393→    expiresAt: new Date(now + ttlMs).toISOString(),
   394→    policyVersion: 0, // filled by the caller (persistSnapshot path)
   395→  };
   396→}

   398→function shapeSnapshot(row: {
   399→  id: string;
   400→  version: number;
   401→  status: string;
   402→  score: number;
   403→  confidence: number;
   404→  riskBand: string;
   405→  components: string;
   406→  explanation: string;
   407→  trigger: string;
   408→  computedAt: Date;
   409→  expiresAt: Date;
   410→  state?: string | null;
   411→  policyId?: string | null;
   412→  frozenAt?: Date | null;
   413→  frozenReason?: string | null;
   414→}) {
   415→  const state =
   416→    row.state === "FROZEN"
   417→      ? "FROZEN"
   418→      : row.state === "RETIRED"
   419→        ? "RETIRED"
   420→        : row.expiresAt.getTime() > Date.now()
   421→          ? "ACTIVE"
   422→          : "STALE";
   423→  return {
   424→    id: row.id,
   425→    version: row.version,
   426→    status: row.status as ScoreStatus,
   427→    score: row.score,
   428→    confidence: row.confidence,
   429→    riskBand: row.riskBand as "LOW" | "MEDIUM" | "HIGH",
   430→    components: JSON.parse(row.components) as ScoreComponent[],
   431→    explanation: JSON.parse(row.explanation) as string[],
   432→    trigger: row.trigger,
   433→    computedAt: row.computedAt.toISOString(),
   434→    expiresAt: row.expiresAt.toISOString(),
   435→    fresh: row.expiresAt.getTime() > Date.now(),
   436→    // Stage 8 — lifecycle + policy provenance
   437→    state,
   438→    policyId: row.policyId ?? null,
   439→    frozenAt: row.frozenAt?.toISOString() ?? null,
   440→    frozenReason: row.frozenReason ?? null,
   441→  };
   442→}

   444→async function persistSnapshot(userId: string, computed: ComputedScore, policyId: string | null) {
   445→  await db.trustScoreSnapshot.create({
   446→    data: {
   447→      userId,
   448→      version: SCORE_VERSION,
   449→      status: computed.status,
   450→      score: computed.score,
   451→      confidence: computed.confidence,
   452→      riskBand: computed.riskBand,
   453→      components: JSON.stringify(computed.components),
   454→      explanation: JSON.stringify(computed.explanation),
   455→      inputsHash: computed.inputsHash,
   456→      trigger: computed.trigger,
   457→      expiresAt: new Date(computed.expiresAt),
   458→      state: "ACTIVE",
   459→      policyId,
   460→    },
   461→  });
   462→  // Prune: keep the latest SNAPSHOT_RETAIN snapshots per user.
   463→  const stale = await db.trustScoreSnapshot.findMany({
   464→    where: { userId },
   465→    orderBy: { computedAt: "desc" },
   466→    skip: SNAPSHOT_RETAIN,
   467→    select: { id: true },
   468→  });
   469→  if (stale.length > 0) {
   470→    await db.trustScoreSnapshot.deleteMany({ where: { id: { in: stale.map((s) => s.id) } } });
   471→  }
   472→}

   474→export async function getScoreSnapshot(userId: string): Promise<ReturnType<typeof shapeSnapshot>> {
   475→  const latest = await db.trustScoreSnapshot.findFirst({
   476→    where: { userId },
   477→    orderBy: { computedAt: "desc" },
   478→  });
   479→
   480→  // Stage 8 — FREEZE (appeal pending): serve the frozen row verbatim.
   481→  // The score cannot move — up or down — until the human decision lands.
   482→  if (latest && latest.state === "FROZEN") {
   483→    return shapeSnapshot(latest);
   484→  }
   485→
   486→  // Stage 8 — the ACTIVE policy drives the numbers AND the inputsHash:
   487→  // a policy change recomputes every user (new snapshot names the policy).
   488→  const policy = await getActivePolicy();
   489→  const rules = policy?.rules;
   490→  if (!rules) {
   491→    // Unreachable in practice (seed guarantees v1) — guard for type safety.
   492→    if (latest) return shapeSnapshot(latest);
   493→    throw new Error("No active scoring policy");
   494→  }
   495→  const ttlMs = rules.snapshotTtlHours * 60 * 60 * 1000;
   496→  const inputs = await collectInputs(userId, rules.interactionWindowDays);
   497→  const hash = sha256Hex(inputsHashFor(inputs) + policyRulesHash(rules));
   498→
   499→  if (
   500→    latest &&
   501→    latest.inputsHash === hash &&
   502→    latest.expiresAt.getTime() > Date.now() &&
   503→    latest.policyId === policy.id
   504→  ) {
   505→    return shapeSnapshot(latest);
   506→  }
   507→
   508→  const trigger =
   509→    latest === undefined || latest === null
   510→      ? "INITIAL"
   511→      : latest.inputsHash !== hash || latest.policyId !== policy.id
   512→        ? "MATERIAL_CHANGE"
   513→        : "PERIODIC";
   514→  const computed = computeScoreFromInputs(inputs, trigger, rules, ttlMs);
   515→  computed.policyVersion = policy.version;
   516→  await persistSnapshot(userId, computed, policy.id);
   517→  const fresh = await db.trustScoreSnapshot.findFirst({
   518→    where: { userId },
   519→    orderBy: { computedAt: "desc" },
   520→  });
   521→  return shapeSnapshot(fresh ?? latest!);
   522→}

   524→// Material-change hook: called by identity/signal services after mutations
   525→// that move the read model. Syncs credentials (so lapsed sources drop out)
   526→// and recomputes the snapshot eagerly — the passport is never stale right
   527→// after an action the user just took.
   528→export async function markMaterialChange(
   529→  userId: string,
   530→  _trigger: string
   531→): Promise<void> {
   532→  await syncCredentials(userId);
   533→  await getScoreSnapshot(userId);
   534→}

   536→// ---------------------------------------------------------------------------
   537→// Credential sync — credentials are platform-issued assertions that shadow
   538→// their source evidence. A credential NEVER outlives its source; a manual
   539→// user revocation sticks even if the source is still live.
   540→// ---------------------------------------------------------------------------
   541→
   542→const CREDENTIAL_LABELS: Record<string, string> = {
   543→  GOV_ID_VERIFIED: "Government identity verified",
   544→  PHONE_VERIFIED: "Phone number verified",
   545→  LIVENESS_VERIFIED: "Biometric liveness passed",
   546→};
   547→
   548→export function credentialLabel(type: string): string {
   549→  return CREDENTIAL_LABELS[type] ?? type;
   550→}
   551→
   552→export async function syncCredentials(userId: string): Promise<number> {
   553→  const identity = await db.trustIdentity.findUnique({
   554→    where: { userId },
   555→    include: {
   556→      identifiers: true,
   557→      attributes: true,
   558→      evidence: { where: { status: "ACTIVE" }, orderBy: { collectedAt: "desc" } },
   559→    },
   560→  });
   561→  if (!identity) return 0;
   562→
   563→  const now = Date.now();
   564→  const fresh = (d: Date | null) => (d?.getTime() ?? 0) > now;
   565→  const govEvidence = identity.evidence.find((e) => e.type === "NINAUTH_ID_TOKEN" && fresh(e.expiresAt));
   566→  const phoneIdentifier = identity.identifiers.find(
   567→    (i) => i.type === "PHONE" && i.status === "ACTIVE" && fresh(i.expiresAt)
   568→  );
   569→  const biometricIdentifier = identity.identifiers.find(
   570→    (i) => i.type === "BIOMETRIC" && i.status === "ACTIVE" && fresh(i.expiresAt)
   571→  );
   572→  const phoneEvidence = identity.evidence.find((e) => e.type === "PHONE_OTP" && fresh(e.expiresAt));
   573→  const livenessEvidence = identity.evidence.find((e) => e.type === "LIVENESS" && fresh(e.expiresAt));
   574→
   575→  const sources: {
   576→    type: string;
   577→    live: boolean;
   578→    evidenceId: string | null;
   579→    expiresAt: Date | null;
   580→    claims: Record<string, unknown>;
   581→  }[] = [
   582→    {
   583→      type: "GOV_ID_VERIFIED",
   584→      live: identity.status === "VERIFIED" && !!govEvidence,
   585→      evidenceId: govEvidence?.id ?? null,
   586→      expiresAt: identity.expiresAt,
   587→      claims: {
   588→        provider: identity.provider,
   589→        providerMode: identity.providerMode,
   590→        maskedRef: identity.providerIdentityRef,
   591→        attributesAsserted: identity.attributes.filter((a) => a.status === "ACTIVE").length,
   592→      },
   593→    },
   594→    {
   595→      type: "PHONE_VERIFIED",
   596→      live: !!phoneIdentifier,
   597→      evidenceId: phoneEvidence?.id ?? null,
   598→      expiresAt: phoneIdentifier?.expiresAt ?? null,
   599→      claims: {
   600→        phoneHint: phoneIdentifier?.hint ?? null,
   601→        simSwapRisk: "checked",
   602→      },
   603→    },
   604→    {
   605→      type: "LIVENESS_VERIFIED",
   606→      live: !!biometricIdentifier,
   607→      evidenceId: livenessEvidence?.id ?? null,
   608→      expiresAt: biometricIdentifier?.expiresAt ?? null,
   609→      claims: {
   610→        verdict: "passed",
   611→        templateStored: false, // honesty: verdict-only, never biometric templates
   612→      },
   613→    },
   614→  ];
   615→
   616→  let changed = 0;
   617→  for (const src of sources) {
   618→    const existing = await db.credential.findUnique({
   619→      where: { userId_type: { userId, type: src.type } },
   620→    });
   621→    if (src.live && (!existing || (existing.status !== "ACTIVE" && !existing.manualRevoked))) {
   622→      await db.credential.upsert({
   623→        where: { userId_type: { userId, type: src.type } },
   624→        create: {
   625→          userId,
   626→          trustIdentityId: identity.id,
   627→          type: src.type,
   628→          issuer: "TrustScore Platform",
   629→          issuerMode: "MOCK",
   630→          claims: JSON.stringify(src.claims),
   631→          status: "ACTIVE",
   632→          evidenceId: src.evidenceId,
   633→          expiresAt: src.expiresAt,
   634→          manualRevoked: false,
   635→        },
   636→        update: {
   637→          status: "ACTIVE",
   638→          claims: JSON.stringify(src.claims),
   639→          evidenceId: src.evidenceId,
   640→          expiresAt: src.expiresAt,
   641→          revokedAt: null,
   642→          trustIdentityId: identity.id,
   643→        },
   644→      });
   645→      changed += 1;
   646→    } else if (!src.live && existing && existing.status === "ACTIVE") {
   647→      // Source gone → credential lapses (non-manual revocation).
   648→      await db.credential.update({
   649→        where: { id: existing.id },
   650→        data: { status: "REVOKED", revokedAt: new Date() },
   651→      });
   652→      changed += 1;
   653→    }
   654→  }
   655→  return changed;
   656→}

   658→export async function getCredentialsForUser(userId: string) {
   659→  const rows = await db.credential.findMany({
   660→    where: { userId },
   661→    orderBy: { issuedAt: "desc" },
   662→  });
   663→  const now = Date.now();
   664→  return rows.map((c) => {
   665→    const expired = (c.expiresAt?.getTime() ?? Infinity) <= now;
   666→    const status =
   667→      c.status === "ACTIVE" && expired ? "EXPIRED" : c.status;
   668→    return {
   669→      id: c.id,
   670→      type: c.type,
   671→      label: credentialLabel(c.type),
   672→      issuer: c.issuer,
   673→      issuerMode: c.issuerMode,
   674→      claims: JSON.parse(c.claims) as Record<string, unknown>,
   675→      status,
   676→      manualRevoked: c.manualRevoked,
   677→      evidenceId: c.evidenceId,
   678→      issuedAt: c.issuedAt.toISOString(),
   679→      expiresAt: c.expiresAt?.toISOString() ?? null,
   680→      revokedAt: c.revokedAt?.toISOString() ?? null,
   681→    };
   682→  });
   683→}
