    85→_windows = {}  # scope -> [window_start_ts, used]
    86→
    87→
    88→def scope_of(path):
    89→    if path in SCOPE_LIMITS:
    90→        return SCOPE_LIMITS[path]
    91→    m = re.match(r"^/api/v1/auth/google/[^/]+/(grant|callback)$", path)
    92→    if m:
    93→        return ("google-grant" if m.group(1) == "grant" else "google-callback", 10)
    94→    return None
    95→
    96→
    97→def pace(path):
    98→    sc = scope_of(path)
    99→    if not sc:
   100→        return
   101→    name, limit = sc
   102→    now = time.time()
   103→    win = _windows.get(name)
   104→    if not win or now - win[0] >= 60:
   105→        win = [now, 0]
   106→        _windows[name] = win
   107→    if win[1] >= limit:
   108→        wait = win[0] + 61 - now
   109→        if wait > 0:
   110→            print(f"  ... pacing {name} rate window ({limit}/min): sleeping {wait:.0f}s")
   111→            time.sleep(wait)
   112→        _windows[name] = [time.time(), 0]
   113→    _windows[name][1] += 1
   114→
   115→
   116→def reset_window(path):
   117→    sc = scope_of(path)
   118→    if sc:
   119→        _windows[sc[0]] = [time.time(), 0]

   122→class Client:
   123→    def __init__(self):
   124→        self.jar = http.cookiejar.CookieJar()
   125→        self.op = urllib.request.build_opener(urllib.request.HTTPCookieProcessor(self.jar))
   126→        self.token = None  # latest raw ts_session token seen in Set-Cookie
   127→
   128→    def _grab_token(self, set_cookies):
   129→        for sc in set_cookies or []:
   130→            if "ts_session=" in sc:
   131→                self.token = sc.split("ts_session=", 1)[1].split(";", 1)[0]
   132→
   133→    def call(self, method, path, body=None):
   134→        pace(path)
   135→        data = json.dumps(body).encode() if body is not None else None
   136→        req = urllib.request.Request(BASE + path, data=data, method=method)
   137→        if data:
   138→            req.add_header("Content-Type", "application/json")
   139→        for attempt in (0, 1, 2):
   140→            try:
   141→                with self.op.open(req, timeout=45) as res:
   142→                    scs = res.headers.get_all("Set-Cookie") or []
   143→                    self._grab_token(scs)
   144→                    return res.status, json.loads(res.read().decode() or "{}"), scs
   145→            except urllib.error.HTTPError as e:
   146→                raw = e.read().decode()
   147→                if e.code == 429 and attempt < 2:
   148→                    print(f"  ... 429 backoff 66s ({path})")
   149→                    time.sleep(66)
   150→                    reset_window(path)
   151→                    continue
   152→                scs = e.headers.get_all("Set-Cookie") or []
   153→                self._grab_token(scs)
   154→                try:
   155→                    b = json.loads(raw or "{}")
   156→                except Exception:
   157→                    b = {}
   158→                return e.code, b, scs
   159→            except Exception as e:
   160→                # Transient dev-server hiccup (lazy compile, OOM restart):
   161→                # brief backoff + retry, then fail the check gracefully.
   162→                if attempt < 2:
   163→                    print(f"  ... connection error on {path} ({type(e).__name__}), retrying")
   164→                    time.sleep(4)
   165→                    continue
   166→                print(f"  ... connection DEAD on {path}: {e}")
   167→                return 0, {"error": {"code": "CONNECTION_FAILED", "message": str(e)}}, []
   168→        return 0, {}, []

   171→def check(name, cond, extra=""):
   172→    global PASS, FAIL
   173→    ok = bool(cond)
   174→    mark = "PASS" if ok else "FAIL"
   175→    if ok:
   176→        PASS += 1
   177→    else:
   178→        FAIL += 1
   179→    print(f"  [{mark}] {name}" + (f" — {extra}" if (extra and not ok) else ""))
   180→    return ok
