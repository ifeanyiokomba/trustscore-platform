   100→function policyDraftTemplate(active: PolicyRulesInfo | undefined): Partial<PolicyRulesInfo> {
   101→  if (!active) return {};
   102→  return {
   103→    credentialPoints: active.credentialPoints,
   104→    credentialMax: active.credentialMax,
   105→    interactionPoints: active.interactionPoints,
   106→    interactionMax: active.interactionMax,
   107→    clearedPoints: active.clearedPoints,
   108→    clearedMax: active.clearedMax,
   109→    riskPenaltyPer: active.riskPenaltyPer,
   110→    riskMaxPenalty: active.riskMaxPenalty,
   111→    snapshotTtlHours: active.snapshotTtlHours,
   112→  };
   113→}

   116→  overview,
   117→  onChanged,
   118→}: {
   119→  overview: EngineAdminOverview;
   120→  onChanged: () => Promise<void>;
   121→}) {
   122→  const { toast } = useToast();
   123→  const active = overview.policies.find((p) => p.status === "ACTIVE");
   124→  const [draftOpen, setDraftOpen] = React.useState(false);
   125→  const [dpiaOpen, setDpiaOpen] = React.useState(false);
   126→  const [gateConfirm, setGateConfirm] = React.useState("");
   127→  const [busy, setBusy] = React.useState<string | null>(null);
   128→
   129→  // Draft form state
   130→  const [rules, setRules] = React.useState<Partial<PolicyRulesInfo>>(() =>
   131→    policyDraftTemplate(active?.rules)
   132→  );
   133→  const [summary, setSummary] = React.useState("");
   134→
   135→  // DPIA form state
   136→  const [dpiaPolicyId, setDpiaPolicyId] = React.useState<string>("");
   137→  const [dpiaSummary, setDpiaSummary] = React.useState("");
   138→  const [dpiaResidual, setDpiaResidual] = React.useState<"LOW" | "MEDIUM" | "HIGH">("LOW");
   139→  const [checks, setChecks] = React.useState<Record<string, boolean>>({});
   140→
   141→  const drafts = overview.policies.filter((p) => p.status === "DRAFT");
   142→  const draftForDpia = dpiaPolicyId
   143→    ? overview.policies.find((p) => p.id === dpiaPolicyId)
   144→    : drafts[0];
   145→
   146→  const budgetTotal =
   147→    (active?.rules.assuranceBase[4] ?? 60) +
   148→    (Number(rules.credentialMax) || 0) +
   149→    (Number(rules.interactionMax) || 0) * (Number(rules.interactionPoints) || 0) +
   150→    (Number(rules.clearedMax) || 0) * (Number(rules.clearedPoints) || 0);
   151→
   152→  async function api(
   153→    name: string,
   154→    path: string,
   155→    body: object,
   156→    method = "POST"
   157→  ): Promise<{ ok: boolean; message: string }> {
   158→    setBusy(name);
   159→    try {
   160→      const res = await fetch(path, {
   161→        method,
   162→        headers: { "Content-Type": "application/json" },
   163→        body: JSON.stringify(body),
   164→      });
   165→      const data = (await res.json().catch(() => ({}))) as {
   166→        note?: string;
   167→        error?: { message?: string };
   168→      };
   169→      const ok = res.ok;
   170→      toast({
   171→        title: ok ? "Done" : "Rejected",
   172→        description: ok ? (data.note ?? "OK") : (data.error?.message ?? "Request failed"),
   173→        variant: ok ? "default" : "destructive",
   174→      });
   175→      if (ok) await onChanged();
   176→      return { ok, message: data.note ?? data.error?.message ?? "" };
   177→    } catch {
   178→      toast({ title: "Error", description: "Network error", variant: "destructive" });
   179→      return { ok: false, message: "network" };
   180→    } finally {
   181→      setBusy(null);
   182→    }
   183→  }
   184→
   185→  return (
   186→    <Card className="ts-card-hover min-w-0" data-testid="engine-admin">
   187→      <CardHeader className="flex-row flex-wrap items-center gap-3 space-y-0">
   188→        <span className="flex h-11 w-11 shrink-0 items-center justify-center rounded-xl bg-primary/10 text-primary">
   189→          <Settings2 className="h-5 w-5" />
   190→        </span>
   191→        <div className="min-w-0 flex-1">
   192→          <CardTitle className="text-base">Engine administration</CardTitle>
   193→          <CardDescription className="truncate">
   194→            Policy versions, DPIA records and the automated-decision gate —
   195→            operational grant (ADMIN), never self-service
   196→          </CardDescription>
   197→        </div>
   198→        <Dialog open={draftOpen} onOpenChange={setDraftOpen}>
   199→          <DialogTrigger asChild>
   200→            <Button
   201→              className="gap-1.5"
   202→              onClick={() => {
   203→                setRules(policyDraftTemplate(active?.rules));
   204→                setSummary("");
   205→              }}
   206→            >
   207→              <Plus className="h-4 w-4" aria-hidden="true" />
   208→              New policy draft
   209→            </Button>
   210→          </DialogTrigger>
   211→          <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-2xl">
   212→            <DialogHeader>
   213→              <DialogTitle>Draft a new scoring policy</DialogTitle>
   214→              <DialogDescription>
   215→                Rules are validated and clamped server-side; the draft activates
   216→                only after a completed DPIA record. Published change summaries
   217→                must be honest and specific.
   218→              </DialogDescription>
   219→            </DialogHeader>
   220→            <div className="space-y-4">
   221→              <div className="grid gap-3 sm:grid-cols-2">
   222→                {RULE_FIELDS.map((f) => (
   223→                  <div key={f.key} className="space-y-1.5">
   224→                    <Label htmlFor={`rule-${f.key}`} className="text-xs">
   225→                      {f.label}{" "}
   226→                      <span className="font-normal text-muted-foreground">({f.hint})</span>
   227→                    </Label>
   228→                    <Input
   229→                      id={`rule-${f.key}`}
   230→                      type="number"
   231→                      min={f.min}
   232→                      max={f.max}
   233→                      step={f.step}
   234→                      value={(rules[f.key] as number) ?? ""}
   235→                      onChange={(e) =>
   236→                        setRules((r) => ({ ...r, [f.key]: Number(e.target.value) }))
   237→                      }
   238→                      className="tabular-nums"
   239→                    />
   240→                  </div>
   241→                ))}
   242→              </div>
   243→              <div className="ts-inset rounded-lg px-3.5 py-3">
   244→                <p className="text-xs font-semibold">
   245→                  Projected budget:{" "}
   246→                  <span
   247→                    className={cn(
   248→                      "tabular-nums",
   249→                      budgetTotal > 100 ? "text-destructive" : "text-primary"
   250→                    )}
   251→                  >
   252→                    {budgetTotal}/100
   253→                  </span>{" "}
   254→                  (identity {active?.rules.assuranceBase[4] ?? 60} + credentials{" "}
   255→                  {Number(rules.credentialMax) || 0} + reputation{" "}
   256→                  {(Number(rules.interactionMax) || 0) * (Number(rules.interactionPoints) || 0)} +
   257→                  resolution{" "}
   258→                  {(Number(rules.clearedMax) || 0) * (Number(rules.clearedPoints) || 0)})
   259→                </p>
   260→                <p className="mt-1 text-[11px] leading-snug text-muted-foreground">
   261→                  {budgetTotal > 100
   262→                    ? "Over 100 — the engine clamps final scores to 0–100, but the server will warn and you should re-balance."
   263→                    : "The engine clamps every rule at write time; snapshots name the policy that produced them."}
   264→                </p>
   265→              </div>
   266→              <div className="space-y-1.5">
   267→                <Label htmlFor="policy-summary" className="text-xs">

   286→                onClick={async () => {
   287→                  const r = await api("draft", "/api/v1/engine/admin/policies", {
   288→                    rules,
   289→                    changeSummary: summary,
   290→                  });
   291→                  if (r.ok) setDraftOpen(false);
   292→                }}
   293→              >
   294→                {busy === "draft" ? (
   295→                  <Loader2 className="mr-2 h-4 w-4 animate-spin" aria-hidden="true" />
   296→                ) : null}
   297→                Create draft
   298→              </Button>
   299→            </DialogFooter>
   300→          </DialogContent>
   301→        </Dialog>
   302→      </CardHeader>
   303→      <CardContent className="space-y-5">
   304→        {/* Snapshot state distribution */}
   305→        <div className="ts-inset rounded-lg px-3.5 py-3.5">
   306→          <p className="flex items-center gap-2 text-xs font-semibold">
   307→            <GitBranch className="h-3.5 w-3.5 text-primary" aria-hidden="true" />
   308→            Snapshot lifecycle distribution
   309→          </p>
   310→          <div className="mt-2.5 flex flex-wrap gap-2">
   311→            {(["ACTIVE", "STALE", "FROZEN", "RETIRED"] as const).map((s) => (
   312→              <span
   313→                key={s}
   314→                className={cn(
   315→                  "rounded-md border px-2 py-1 text-[11px] font-semibold tabular-nums",
   316→                  s === "FROZEN"
   317→                    ? "border-teal-600/40 bg-teal-600/10 text-teal-700 dark:text-teal-300"
   318→                    : "border-border bg-muted/40 text-muted-foreground"
   319→                )}
   320→              >
   321→                {s} · {overview.snapshots.byState[s] ?? 0}
   322→              </span>
   323→            ))}
   324→            <span className="rounded-md border border-border px-2 py-1 text-[11px] font-semibold tabular-nums text-muted-foreground">
   325→              total · {overview.snapshots.total}
   326→            </span>
   327→          </div>
   328→        </div>
   329→
   330→        {/* Policy versions */}
   331→        <div>
   332→          <p className="flex items-center gap-2 text-xs font-semibold">

   336→          <ul className="mt-2.5 space-y-2.5">
   337→            {overview.policies.map((p) => (
   338→              <motion.li
   339→                key={p.id}
   340→                initial={{ opacity: 0, y: 6 }}
   341→                animate={{ opacity: 1, y: 0 }}
   342→                className={cn(
   343→                  "rounded-lg border px-3.5 py-3",
   344→                  p.status === "ACTIVE" ? "border-primary/40 bg-primary/5" : "border-border"
   345→                )}
   346→              >
   347→                <div className="flex flex-wrap items-center gap-2">
   348→                  <span className="font-mono text-xs font-bold">v{p.version}</span>
   349→                  <Badge
   350→                    variant="outline"
   351→                    className={cn(
   352→                      "text-[10px] font-semibold",
   353→                      p.status === "ACTIVE"
   354→                        ? "bg-primary/15 text-primary"
   355→                        : p.status === "DRAFT"
   356→                          ? "bg-amber-500/10 text-amber-600 dark:text-amber-400"
   357→                          : "bg-muted text-muted-foreground"
   358→                    )}
   359→                  >
   360→                    {p.status}
   361→                  </Badge>
   362→                  <span className="ml-auto text-[10px] text-muted-foreground">
   363→                    {p.status === "DRAFT"
   364→                      ? "awaiting DPIA + activation"
   365→                      : p.activatedAt
   366→                        ? `activated ${new Date(p.activatedAt).toLocaleDateString("en-NG", { day: "numeric", month: "short", year: "numeric" })}`
   367→                        : ""}
   368→                  </span>
   369→                </div>
   370→                <p className="mt-1.5 line-clamp-2 text-[11px] leading-relaxed text-muted-foreground">
   371→                  {p.changeSummary}
   372→                </p>
   373→                {p.status === "DRAFT" ? (
   374→                  <div className="mt-2.5 flex flex-wrap gap-2">
   375→                    <Button
   376→                      size="sm"
   377→                      variant="outline"
   378→                      className="h-8 gap-1.5"
   379→                      onClick={() => {
   380→                        setDpiaPolicyId(p.id);
   381→                        setDpiaSummary("");
   382→                        setChecks({});
   383→                        setDpiaOpen(true);
   384→                      }}
   385→                    >
   386→                      <FileCheck2 className="h-3.5 w-3.5" aria-hidden="true" />

   416→                        </AlertDialogFooter>
   417→                      </AlertDialogContent>

   421→              </motion.li>

   440→                    <Badge
   441→                      variant="outline"

   461→                      </span>
